cmake_minimum_required(VERSION 3.21)
project(gt86_firmware_research LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(firmware_reader src/firmware.cpp)
target_include_directories(firmware_reader PUBLIC src)
add_executable(fwinspect src/main.cpp)
target_link_libraries(fwinspect PRIVATE firmware_reader)
add_library(minilzo STATIC third_party/lzo-2.10/minilzo/minilzo.c)
target_include_directories(minilzo PUBLIC third_party/lzo-2.10/minilzo third_party/lzo-2.10/include/lzo)
add_library(qnx_reader src/qnx.cpp)
target_link_libraries(qnx_reader PUBLIC firmware_reader PRIVATE minilzo)
add_executable(qnxinspect src/qnx_main.cpp)
target_link_libraries(qnxinspect PRIVATE qnx_reader)
if(MSVC)
  target_compile_options(firmware_reader PRIVATE /W4 /permissive-)
  target_compile_options(fwinspect PRIVATE /W4 /permissive-)
  target_compile_options(qnx_reader PRIVATE /W4 /permissive-)
  target_compile_options(qnxinspect PRIVATE /W4 /permissive-)
endif()
enable_testing()
add_library(carplay_protocol STATIC src/carplay/iap2_wire.c src/carplay/iap2_auth.c src/carplay/iap2_link.c src/carplay/iap2_control.c src/carplay/iap2_identification.c src/carplay/iap2_transport.c src/carplay/iap2_carplay.c src/carplay/iap2_power.c src/carplay/usbmux_wire.c src/carplay/usbmux_host.c src/carplay/usbmux_connection.c)
set_target_properties(carplay_protocol PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_include_directories(carplay_protocol PUBLIC src/carplay)
if(MSVC)
  target_compile_options(carplay_protocol PRIVATE /W4)
else()
  target_compile_options(carplay_protocol PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(iap2_tests tests/iap2_tests.cpp)
target_link_libraries(iap2_tests PRIVATE carplay_protocol)
add_test(NAME iap2_tests COMMAND iap2_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/iap2-csm-vectors.txt")
add_executable(iap2_link_tests tests/iap2_link_tests.cpp)
target_link_libraries(iap2_link_tests PRIVATE carplay_protocol)
add_test(NAME iap2_link_tests COMMAND iap2_link_tests)
add_executable(iap2_control_tests tests/iap2_control_tests.cpp)
target_link_libraries(iap2_control_tests PRIVATE carplay_protocol)
add_test(NAME iap2_control_tests COMMAND iap2_control_tests)
add_executable(iap2_identification_tests tests/iap2_identification_tests.cpp)
target_link_libraries(iap2_identification_tests PRIVATE carplay_protocol)
add_test(NAME iap2_identification_tests COMMAND iap2_identification_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/iap2-csm-vectors.txt")
add_executable(iap2_transport_tests tests/iap2_transport_tests.cpp)
target_link_libraries(iap2_transport_tests PRIVATE carplay_protocol)
add_test(NAME iap2_transport_tests COMMAND iap2_transport_tests)
add_executable(iap2_carplay_tests tests/iap2_carplay_tests.cpp)
target_link_libraries(iap2_carplay_tests PRIVATE carplay_protocol)
add_test(NAME iap2_carplay_tests COMMAND iap2_carplay_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/iap2-csm-vectors.txt")
add_executable(usbmux_tests tests/usbmux_tests.cpp)
target_link_libraries(usbmux_tests PRIVATE carplay_protocol)
add_test(NAME usbmux_tests COMMAND usbmux_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/usbmux-wire-vectors.txt")
add_executable(usbmux_host_tests tests/usbmux_host_tests.cpp)
target_link_libraries(usbmux_host_tests PRIVATE carplay_protocol)
add_test(NAME usbmux_host_tests COMMAND usbmux_host_tests)
add_executable(usbmux_connection_tests tests/usbmux_connection_tests.cpp)
target_link_libraries(usbmux_connection_tests PRIVATE carplay_protocol)
add_test(NAME usbmux_connection_tests COMMAND usbmux_connection_tests)
add_executable(reader_tests tests/reader_tests.cpp)
target_link_libraries(reader_tests PRIVATE firmware_reader)
add_test(NAME reader_tests COMMAND reader_tests)
add_executable(qnx_tests tests/qnx_tests.cpp)
target_link_libraries(qnx_tests PRIVATE qnx_reader minilzo)
add_test(NAME qnx_tests COMMAND qnx_tests)
if(WIN32)
  add_test(NAME qnx_extract_tests COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_SOURCE_DIR}/tests/qnx_extract_tests.ps1"
    -Inspector "$<TARGET_FILE:qnxinspect>" -FixtureGenerator "$<TARGET_FILE:qnx_tests>")
  add_test(NAME zip_tests COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_SOURCE_DIR}/tests/zip_tests.ps1" -BuildDirectory "${CMAKE_BINARY_DIR}")
endif()
