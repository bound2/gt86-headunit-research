cmake_minimum_required(VERSION 3.21)
project(lua_research_host LANGUAGES C)
if(NOT EXISTS "${LUA_SOURCE_DIR}/lua.h")
  message(FATAL_ERROR "Set LUA_SOURCE_DIR to the extracted, verified Lua 5.1.5 src directory")
endif()
file(GLOB LUA_FILES "${LUA_SOURCE_DIR}/*.c")
list(REMOVE_ITEM LUA_FILES "${LUA_SOURCE_DIR}/lua.c" "${LUA_SOURCE_DIR}/luac.c" "${LUA_SOURCE_DIR}/print.c")
add_library(lua51 STATIC ${LUA_FILES})
target_compile_definitions(lua51 PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(lua51 PUBLIC "${LUA_SOURCE_DIR}")
add_executable(lua "${LUA_SOURCE_DIR}/lua.c")
target_link_libraries(lua PRIVATE lua51)
add_executable(luac "${LUA_SOURCE_DIR}/luac.c" "${LUA_SOURCE_DIR}/print.c")
target_link_libraries(luac PRIVATE lua51)
